cmake_minimum_required(VERSION 3.24)
project(osg_template_tests LANGUAGES C)
include(FetchContent)
include(../cmake/common-revision.cmake)
FetchContent_Declare(firmware_common GIT_REPOSITORY https://github.com/opensimgear/firmware-common.git
  GIT_TAG ${OSG_COMMON_REVISION} GIT_SUBMODULES "")
set(OSG_EMBEDDED OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(firmware_common)
include(${firmware_common_SOURCE_DIR}/cmake/dependencies.cmake)
set(OSG_CYPHAL_BUILD_TESTS ON)
add_subdirectory(../shared-cyphal shared-cyphal)
enable_testing()
add_executable(console_info_test console_info_test.c ../src/console_info.c)
target_include_directories(console_info_test PRIVATE ../include)
target_compile_features(console_info_test PRIVATE c_std_11)
target_compile_options(console_info_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME complete_usb_identity_response COMMAND console_info_test)
add_executable(runtime_config_test runtime_config_test.c ../src/runtime_config.c ../shared-gateway/src/gateway_journal.c)
target_include_directories(runtime_config_test PRIVATE ../include ../shared-gateway/include)
target_link_libraries(runtime_config_test PRIVATE osg_core osg_cyphal)
target_compile_features(runtime_config_test PRIVATE c_std_11)
target_compile_options(runtime_config_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME runtime_configuration COMMAND runtime_config_test)
add_executable(device_actions_test device_actions_test.c ../src/device_actions.c ../src/config_can.c ../src/can_client.c
  ../src/runtime_config.c ../shared-gateway/src/gateway_journal.c)
target_include_directories(device_actions_test PRIVATE ../include ../shared-gateway/include)
target_link_libraries(device_actions_test PRIVATE osg_core osg_cyphal)
target_compile_features(device_actions_test PRIVATE c_std_11)
target_compile_options(device_actions_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME advertised_device_actions COMMAND device_actions_test)
add_executable(hid_delivery_test hid_delivery_test.c)
target_include_directories(hid_delivery_test PRIVATE ../include)
target_compile_features(hid_delivery_test PRIVATE c_std_11)
target_compile_options(hid_delivery_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME hid_delivery_recovery COMMAND hid_delivery_test)
foreach(profile example shifter expanded master_caution radio_panel output_only)
  add_executable(profile_${profile} profile_test.c ../src/profile.c ../src/can_delivery.c ../src/runtime_config.c ../shared-gateway/src/gateway_journal.c)
  target_include_directories(profile_${profile} PRIVATE ../include ../profiles ../shared-gateway/include)
  target_link_libraries(profile_${profile} PRIVATE osg_core osg_cyphal)
  add_test(NAME profile_${profile}_behavior COMMAND profile_${profile})
  add_executable(hid_${profile} hid_test.c ../src/profile.c ../src/usb_descriptors.c ../src/runtime_config.c ../shared-gateway/src/gateway_journal.c ${firmware_common_SOURCE_DIR}/src/usb/descriptor.c)
  target_include_directories(hid_${profile} PRIVATE usb ../include ../profiles ../boards/starter_g0b1cb_v0_1_0
    ${firmware_common_SOURCE_DIR}/include ${tinyusb_SOURCE_DIR}/src)
  target_include_directories(hid_${profile} PRIVATE ../shared-gateway/include)
  target_link_libraries(hid_${profile} PRIVATE osg_core osg_cyphal)
  # Compile-only test identifiers; never assigned or flashed.
  target_compile_definitions(hid_${profile} PRIVATE OSG_USB_VID=0xFFFF OSG_USB_PID=0xFFFF)
  add_test(NAME hid_${profile}_contract COMMAND hid_${profile})
  foreach(target profile_${profile} hid_${profile})
    target_compile_definitions(${target} PRIVATE OSG_DEVICE_PROFILE_HEADER="${profile}.h")
    target_compile_features(${target} PRIVATE c_std_11)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  endforeach()
endforeach()
add_executable(can_client_test can_client_test.c ../src/can_client.c ../src/can_delivery.c ../src/boot_session.c
  ../shared-gateway/src/gateway_journal.c)
target_include_directories(can_client_test PRIVATE ../include ../shared-gateway/include)
target_link_libraries(can_client_test PRIVATE osg_core osg_cyphal)
target_compile_features(can_client_test PRIVATE c_std_11)
target_compile_options(can_client_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME can_client_delivery_and_boot COMMAND can_client_test)
