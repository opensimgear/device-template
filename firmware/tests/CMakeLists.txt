cmake_minimum_required(VERSION 3.24)
project(osg_template_tests LANGUAGES C)
include(FetchContent)
include(../cmake/common-revision.cmake)
FetchContent_Declare(firmware_common GIT_REPOSITORY https://github.com/opensimgear/firmware-common.git
  GIT_TAG ${OSG_COMMON_REVISION} GIT_SUBMODULES "")
set(OSG_EMBEDDED OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(firmware_common)
include(${firmware_common_SOURCE_DIR}/cmake/dependencies.cmake)
enable_testing()
add_executable(profile_test profile_test.c ../src/profile.c)
target_include_directories(profile_test PRIVATE ../include)
target_link_libraries(profile_test PRIVATE osg_core)
add_test(NAME template_input_behavior COMMAND profile_test)
add_executable(hid_test hid_test.c ../src/usb_descriptors.c ${firmware_common_SOURCE_DIR}/src/usb/descriptor.c)
target_include_directories(hid_test PRIVATE usb ../include ../boards/starter_g0b1cb
  ${firmware_common_SOURCE_DIR}/include ${tinyusb_SOURCE_DIR}/src)
# Compile-only test identifiers; never assigned or flashed.
target_compile_definitions(hid_test PRIVATE OSG_USB_VID=0xFFFF OSG_USB_PID=0xFFFF)
add_test(NAME template_hid_contract COMMAND hid_test)
foreach(target profile_test hid_test)
  target_compile_features(${target} PRIVATE c_std_11)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endforeach()
